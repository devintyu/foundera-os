'use client'

import { useState } from 'react'
import { Button } from '@/components/ui/button'
import { Zap } from 'lucide-react'

const topUpPackages = [
  { 
    id: 'small',
    credits: 10, 
    priceUSD: 10,
    priceMYR: 46,
    priceId: process.env.NEXT_PUBLIC_STRIPE_TOPUP_SMALL_PRICE_ID,
    popular: false,
    savings: null,
  },
  { 
    id: 'medium',
    credits: 30, 
    priceUSD: 29,
    priceMYR: 135,
    priceId: process.env.NEXT_PUBLIC_STRIPE_TOPUP_MEDIUM_PRICE_ID,
    popular: true,
    savings: '3% off',
  },
  { 
    id: 'large',
    credits: 100, 
    priceUSD: 79,
    priceMYR: 369,
    priceId: process.env.NEXT_PUBLIC_STRIPE_TOPUP_LARGE_PRICE_ID,
    popular: false,
    savings: '21% off',
  },
  { 
    id: 'premium',
    credits: 300, 
    priceUSD: 199,
    priceMYR: 929,
    priceId: process.env.NEXT_PUBLIC_STRIPE_TOPUP_PREMIUM_PRICE_ID,
    popular: false,
    savings: '34% off',
  },
]

export default function TopUpOptions() {
  const [loading, setLoading] = useState<string | null>(null)

  const handleTopUp = async (priceId: string, credits: number) => {
    if (!priceId) {
      alert('Price ID not configured. Please check environment variables.')
      return
    }

    setLoading(priceId)
    
    try {
      const response = await fetch('/api/stripe/topup', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ priceId, credits }),
      })

      const data = await response.json()
      
      if (data.url) {
        window.location.href = data.url
      } else {
        alert(data.error || 'Failed to initiate top-up. Please try again.')
      }
    } catch (error) {
      console.error('Top-up error:', error)
      alert('Failed to initiate top-up. Please try again.')
    } finally {
      setLoading(null)
    }
  }

  return (
    <div>
      <h3 className="text-xl font-semibold text-white mb-4">Top Up Credits</h3>
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
        {topUpPackages.map((pkg) => (
          <div
            key={pkg.id}
            className={`
              bg-[#12121A] border rounded-lg p-6 relative transition-all hover:scale-105
              ${pkg.popular ? 'border-cyan-400 shadow-lg shadow-cyan-400/20' : 'border-white/10'}
            `}
          >
            {pkg.popular && (
              <div className="absolute -top-3 left-1/2 -translate-x-1/2 bg-cyan-400 text-black text-xs font-bold px-3 py-1 rounded-full">
                MOST POPULAR
              </div>
            )}
            
            {pkg.savings && (
              <div className="absolute top-3 right-3 bg-green-500/20 text-green-400 text-xs font-bold px-2 py-1 rounded">
                {pkg.savings}
              </div>
            )}
            
            <div className="text-center mb-4">
              <Zap className="w-10 h-10 text-cyan-400 mx-auto mb-3" />
              <h4 className="text-3xl font-bold text-white mb-1">{pkg.credits}</h4>
              <p className="text-sm text-gray-400">Credits</p>
              
              <div className="mt-4">
                <p className="text-3xl font-bold text-cyan-400">
                  ${pkg.priceUSD}
                </p>
                <p className="text-sm text-gray-400 mt-1">
                  or RM{pkg.priceMYR}
                </p>
              </div>
            </div>

            <Button
              onClick={() => handleTopUp(pkg.priceId!, pkg.credits)}
              disabled={loading === pkg.priceId || !pkg.priceId}
              className={`w-full ${pkg.popular ? 'bg-cyan-400 hover:bg-cyan-500 text-black' : ''}`}
              variant={pkg.popular ? 'default' : 'outline'}
            >
              {loading === pkg.priceId ? (
                <span className="flex items-center gap-2">
                  <div className="w-4 h-4 border-2 border-current border-t-transparent rounded-full animate-spin" />
                  Processing...
                </span>
              ) : (
                'Top Up Now'
              )}
            </Button>
          </div>
        ))}
      </div>
    </div>
  )
}
